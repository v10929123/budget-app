🔗 股價代理伺服器（Proxy）

設定 Cloudflare Workers Proxy URL，以取得台股、美股及貴金屬即時報價（解決 CORS 限制）

📋 Cloudflare Worker 部署說明 
export default {
  async fetch(req) {
    const u=new URL(req.url);
    const type=u.searchParams.get('type');
    const sym=u.searchParams.get('symbol')||'';
    const h={'Access-Control-Allow-Origin':'*','Content-Type':'application/json'};
    try{
      let url,res,data;
      if(type==='tw'){
        url=`https://mis.twse.com.tw/stock/api/getStockInfo.jsp?ex_ch=tse_${sym}.tw&json=1`;
      }else if(type==='us'){
        url=`https://query1.finance.yahoo.com/v8/finance/chart/${sym}?range=1d&interval=1d`;
      }else if(type==='gold'){
        url=`https://data-asg.goldprice.org/GetData/USD-XAU/1`;
      }else if(type==='silver'){
        url=`https://data-asg.goldprice.org/GetData/USD-XAG/1`;
      }else if(type==='fx'){
        url=`https://open.er-api.com/v6/latest/USD`;
      }else return new Response('{}',{headers:h});
      res=await fetch(url,{headers:{'User-Agent':'Mozilla/5.0'}});
      data=await res.text();
      return new Response(data,{headers:h});
    }catch(e){return new Response('{}',{headers:h});}
  }
};

2. 新建 Worker，貼入以上程式碼並部署
3. 複製 Worker URL 填入上方欄位
